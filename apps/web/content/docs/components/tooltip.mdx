## Usage

```tsx
import { Tooltip, TooltipPopup, TooltipTrigger } from "@/components/ui/tooltip";
```

```tsx
<Tooltip>
  <TooltipTrigger render={<Button variant="outline" />}>Hover me</TooltipTrigger>
  <TooltipPopup>Helpful hint</TooltipPopup>
</Tooltip>
```

### Hidden arrow
