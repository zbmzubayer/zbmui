## Usage

```tsx
import { IconSearch } from "@tabler/icons-react";

import { InputGroup, InputGroupAddon, InputGroupInput } from "@/registry/default/ui/input-group";
```

```tsx
<InputGroup>
  <InputGroupInput placeholder="Search..." />
  <InputGroupAddon>
    <IconSearch />
  </InputGroupAddon>
</InputGroup>
```
