## Usage

```tsx
import { Avatar, AvatarFallback, AvatarImage } from "@/components/ui/avatar";
```

```tsx
<Avatar>
  <AvatarImage src="https://github.com/zbmzubayer.png" alt="ZBM Zubayer" />
  <AvatarFallback>ZM</AvatarFallback>
</Avatar>
```
